### 多任务协程返回值

```python
import asyncio

async def faker1():
    print("任务1开始")
    await asyncio.sleep(1)
    print("任务1完成")
    return "任务1结束"


async def faker2():
    print("任务2开始")
    await asyncio.sleep(2)
    print("任务2完成")
    return "任务2结束"


async def faker3():
    print("任务3开始")
    await asyncio.sleep(3)
    print("任务3完成")
    return "任务3结束"


async def main():
    tasks = [
        asyncio.create_task(faker3()),
        asyncio.create_task(faker1()),
        asyncio.create_task(faker2()),
    ]
    # 方案一, 用wait, 返回的结果在result中
    result, pending = await asyncio.wait(tasks)
    for r in result:
        print(r.result())
        
    # 方案二, 用gather, 返回的结果在result中, 结果会按照任务添加的顺序来返回数据
    # 	return_exceptions如果任务在执行过程中报错了. 返回错误信息. 
    result = await asyncio.gather(*tasks, return_exceptions=True)
    for r in result:
        print(r)


if __name__ == '__main__':
    asyncio.run(main())

```

实例代码:

```python
import aiohttp
import asyncio
import aiofiles


async def download(url):
    try:
        name = url.split("/")[-1]
        async with aiohttp.ClientSession() as session:
            # 发送请求, 这里和requests.get()几乎没区别, 除了代理换成了proxy
            async with session.get(url) as resp:
                # # resp.text(encoding='') 这可以设置字符集 
                # 读取数据. 如果想要读取源代码. 直接resp.text()即可. 比原来多了个()
                content = await resp.content.read()
                # 写入文件, 用默认的open也OK. 用aiofiles能进一步提升效率
                async with aiofiles.open(name, mode="wb") as f:
                    await f.write(content)
                    return "OK"
    except:
        print(123)
        return "NO"


async def main():
    url_list = [
        # 所有图片地址都挂了. 自己换下就好, www.xiurenji.vip 老多了. 
        "https://x.u5w.cc/Uploadfile/202110/20/42214426253.jpg",
        "https://x.u5w.cc/Uploadfile/202110/20/B3214426373.jpg",
        "https://www.xiurenji.vip/uploadfile/202110/20/1F214426892.jpg",
        "https://www.xiurenji.vip/uploadfile/202110/20/91214426753.jpg"
    ]
    tasks = []

    for url in url_list:
        # 创建任务
        task = asyncio.create_task(download(url))
        tasks.append(task)

    await asyncio.wait(tasks)


if __name__ == '__main__':
    asyncio.run(main())

```

### 协程案例-扒光一部小说需要多久?

目标, 明朝那些事儿 http://www.sharereading.cc/0/254/

```python
import requests

import aiohttp
import aiofiles

import asyncio
from lxml import etree
from urllib.parse import urljoin


my_headers = {
        "accept": "text/html,application/xhtml+xml,application/xml;q=0.9,image/avif,image/webp,image/apng,*/*;q=0.8,application/signed-exchange;v=b3;q=0.7",
        "accept-encoding": "gzip, deflate",
        "accept-language": "zh-CN,zh;q=0.9,en;q=0.8",
        "cache-control": "no-cache",
        "connection": "close",
        "cookie": "articlevisited=1",
        "dnt": "1",
        "host": "www.sharereading.cc",
        "pragma": "no-cache",
        "referer": "http://www.sharereading.cc/0/254/",
        "upgrade-insecure-requests": "1",
        "user-agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/127.0.0.0 Safari/537.36"
    }

def get_all_chapter_url():
    url = "http://www.sharereading.cc/0/254/"
    session = requests.session()
    session.headers = {
        "accept": "text/html,application/xhtml+xml,application/xml;q=0.9,image/avif,image/webp,image/apng,*/*;q=0.8,application/signed-exchange;v=b3;q=0.7",
        "accept-encoding": "gzip, deflate",
        "accept-language": "zh-CN,zh;q=0.9,en;q=0.8",
        "cache-control": "no-cache",
        "connection": "keep-alive",
        "cookie": "articlevisited=1",
        "dnt": "1",
        "host": "www.sharereading.cc",
        "pragma": "no-cache",
        "referer": "http://www.sharereading.cc/0/254/179492.html",
        "upgrade-insecure-requests": "1",
        "user-agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/127.0.0.0 Safari/537.36"
    }

    resp = session.get(url)
    tree = etree.HTML(resp.text)
    a_list = tree.xpath("//div[@class='info-chapters flex flex-wrap']/a")
    result = []
    for a in a_list:
        href = a.xpath("./@href")[0].strip()
        text = a.xpath("./text()")[0].strip()
        c_url = urljoin(url, href)
        result.append({"name": text, "chapter_url": c_url})
    return result


async def download_one(name, c_url):
    async with aiohttp.ClientSession(headers=my_headers) as session:
        async with session.get(c_url) as resp:
            txt = await resp.text(encoding="utf-8")
            tree = etree.HTML(txt)
            content = tree.xpath("//article[@id='article']//text()")
            content = "".join(content)
            async with aiofiles.open(f"{name}.txt",mode="w", encoding="utf-8") as f:
                await f.write(content)
                print(f"下载{name}完成!")


async def download_all_chapter(chapter_urls):
    tasks = []
    for chapter in chapter_urls:
        name = chapter.get("name")
        c_url = chapter.get("chapter_url")

        t = asyncio.create_task(download_one(name, c_url))
        tasks.append(t)
    await asyncio.wait(tasks)


def main():
    # 1. 获取所有章节的url
    chapter_urls = get_all_chapter_url()
    # 2. 下载这些url. 开始协程
    asyncio.run(download_all_chapter(chapter_urls))


if __name__ == '__main__':
    main()
```
